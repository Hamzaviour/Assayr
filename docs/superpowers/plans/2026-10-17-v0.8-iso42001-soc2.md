# v0.8 ISO/IEC 42001 and SOC 2 Type 2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add ISO/IEC 42001 (AI management system) with an AI systems register and AI impact assessment, and SOC 2 Security with Type 2 tested audits, sharing evidence with ISO 27001.

**Architecture:** Both frameworks reuse the existing catalogue (`Framework`, `Requirement`, `Control`, `ControlRequirement`, `EvidenceSlot`). A new `Control.framework` column separates them from ISO 27001; 42001 control codes carry an `AI ` prefix (the NIA convention) so code lookups stay unique. A new `assayr.ai` package holds the register and impact assessments. Audits gain two frameworks and SOC 2 sample/result fields plus a tracker-layout workbook.

**Tech Stack:** FastAPI, SQLAlchemy 2, Alembic, pytest (backend, `uv`); React 19, TanStack Router/Query, Tailwind, Vitest (frontend); Inno Setup (Windows build).

**Spec:** `docs/superpowers/specs/2026-10-17-v0.8-iso42001-soc2-design.md`

## Global Constraints

- Ship requirement IDs, short headings and Assayr's own summaries only: no ISO text, no AICPA criteria text or points of focus, no text from the user's workbook. Summaries ≤ 400 characters.

- `Control.framework` values: `iso27001`, `nia`, `iso42001`, `soc2`.
- SOC 2 results: `no_exception`, `exception`, `not_tested`. ISO/NIA/42001 results: `pass`, `fail`, `partial`, `na`.

"""ISO/IEC 42001 and SOC 2 catalogues: loading and seeding. Safe on every start: adds only what is missing."""

import json
import uuid
from dataclasses import dataclass
from functools import cache
from pathlib import Path

from assayr.catalogue.models import Control, ControlRequirement, EvidenceSlot, Framework, Requirement
from assayr.identity.models import Scope

DATA = Path(__file__).resolve().parent / "data"
ISO42001_CODE = "ISO42001:2023"
SOC2_CODE = "SOC2:2017"
AI_PREFIX = "AI "


@dataclass
class FrameworkSeed:
    requirements_created: int = 0
    controls_created: int = 0


@cache
def load_iso42001() -> dict:
    return json.loads((DATA / "iso42001_2023.json").read_text(encoding="utf-8"))


@cache
def load_soc2() -> dict:
    return json.loads((DATA / "soc2_tsc_security.json").read_text(encoding="utf-8"))


def control_code(framework: str, ref: str) -> str:
    return AI_PREFIX + ref if framework == "iso42001" else ref


def display_ref(code: str) -> str:
    return code.removeprefix(AI_PREFIX)


@cache
def iso_map() -> dict[str, list[str]]:
    a, s = load_iso42001(), load_soc2()
    out = {control_code("iso42001", c["ref"]): c["iso27001"] for c in a["controls"]}
    out |= {control_code("iso42001", c["ref"]): c["iso27001"] for c in a["clauses"] if c.get("summary")}
    out |= {c["ref"]: c["iso27001"] for c in s["criteria"]}
    return out


def _framework(db: Session, code: str, name: str, version: str) -> Framework:
    fw = db.scalar(select(Framework).where(Framework.code == code))
    if fw is None:
        fw = Framework(id=uuid.uuid4(), code=code, name=name, version=version)
        db.add(fw)
        db.flush()
    return fw


def _seed(db: Session, scope: Scope, framework: str, fw: Framework, groups, leaves) -> FrameworkSeed:
    """groups: (ref, title, parent_ref, kind); leaves: dicts with ref, title, group, theme, summary, evidence_hint."""
    result = FrameworkSeed()
    reqs = {r.ref: r for r in db.scalars(select(Requirement).where(Requirement.framework_id == fw.id))}
    order = 0

    def ensure(ref, title, kind, parent, guidance=None, attributes=None):
        nonlocal order
        order += 1
        if ref not in reqs:
            reqs[ref] = Requirement(
                id=uuid.uuid4(), framework_id=fw.id, ref=ref, title=title, kind=kind,
                parent_id=reqs[parent].id if parent else None, guidance=guidance,
                attributes=attributes or {}, sort_order=order,
            )
            db.add(reqs[ref])
            result.requirements_created += 1
        return reqs[ref]

    for ref, title, parent, kind in groups:
        ensure(ref, title, kind, parent)
    existing = set(db.scalars(select(Control.code).where(Control.scope_id == scope.id)))
    links = []
    for index, leaf in enumerate(leaves, start=1):
        req = ensure(leaf["ref"], leaf["title"], leaf["kind"], leaf["group"], leaf["summary"],
                     {"evidence_hint": leaf["evidence_hint"], "iso27001": leaf["iso27001"], "auto": leaf.get("auto")})
        code = control_code(framework, leaf["ref"])
        if code in existing:
            continue
        control = Control(
            id=uuid.uuid4(), scope_id=scope.id, framework=framework, code=code, title=leaf["title"],
            theme=leaf["theme"], description=leaf["summary"], review_frequency="Annual", sort_order=index,
        )
        control.slots.append(EvidenceSlot(
            id=uuid.uuid4(), name=leaf["evidence_hint"][:300], evidence_type="document", required=True,
            freshness_days=365, rules=[], required_concepts=[], sort_order=1,
        ))
        db.add(control)
        links.append(ControlRequirement(control_id=control.id, requirement_id=req.id))
        result.controls_created += 1
    db.flush()
    db.add_all(links)
    db.flush()
    return result


def seed_iso42001(db: Session, scope: Scope) -> FrameworkSeed:
    data = load_iso42001()
    fw = _framework(db, ISO42001_CODE, data["name"], data["version"])
    titles = {c["ref"]: c["title"] for c in data["clauses"]}
    groups = [(c["ref"], c["title"], c["parent"], "clause_group") for c in data["clauses"] if not c.get("summary")]
    groups += [(o["ref"], o["title"], None, "theme") for o in data["objectives"]]
    objective = {o["ref"]: o["title"] for o in data["objectives"]}
    leaves = [
        {**c, "kind": "clause", "group": c["parent"], "theme": titles[c["ref"].split(".")[0]]}
        for c in data["clauses"] if c.get("summary")
    ]
    leaves += [{**c, "kind": "annex_control", "group": c["objective"], "theme": objective[c["objective"]]}
               for c in data["controls"]]
    return _seed(db, scope, "iso42001", fw, groups, leaves)


def seed_soc2(db: Session, scope: Scope) -> FrameworkSeed:
    data = load_soc2()
    fw = _framework(db, SOC2_CODE, data["name"], data["version"])
    series = {s["ref"]: s["title"] for s in data["series"]}
    groups = [(ref, title, None, "theme") for ref, title in series.items()]
    leaves = [{**c, "kind": "criterion", "group": c["series"], "theme": series[c["series"]]} for c in data["criteria"]]
    return _seed(db, scope, "soc2", fw, groups, leaves)
